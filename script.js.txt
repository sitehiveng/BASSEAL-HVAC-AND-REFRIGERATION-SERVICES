document.addEventListener('DOMContentLoaded', function () {

  /* ---- Loader ---- */
  var loader = document.querySelector('.loader');
  if (loader) {
    window.addEventListener('load', function () {
      setTimeout(function () { loader.classList.add('hide'); }, 260);
    });
    // Fallback in case load already fired
    setTimeout(function () { loader.classList.add('hide'); }, 1800);
  }

  /* ---- Sticky header shrink + active link ---- */
  var header = document.getElementById('siteHeader');
  function onScrollHeader() {
    if (!header) return;
    if (window.scrollY > 12) header.classList.add('scrolled');
    else header.classList.remove('scrolled');
  }
  document.addEventListener('scroll', onScrollHeader, { passive: true });
  onScrollHeader();

  /* ---- Mobile nav toggle ---- */
  var toggle = document.getElementById('navToggle');
  var nav = document.getElementById('mainNav');
  if (toggle && nav) {
    toggle.addEventListener('click', function () {
      toggle.classList.toggle('open');
      nav.classList.toggle('open');
    });
    nav.querySelectorAll('a').forEach(function (a) {
      a.addEventListener('click', function () {
        toggle.classList.remove('open');
        nav.classList.remove('open');
      });
    });
  }

  /* ---- Scroll reveal ---- */
  var revealEls = document.querySelectorAll('.reveal');
  if ('IntersectionObserver' in window && revealEls.length) {
    var io = new IntersectionObserver(function (entries) {
      entries.forEach(function (entry) {
        if (entry.isIntersecting) {
          entry.target.classList.add('in');
          io.unobserve(entry.target);
        }
      });
    }, { threshold: 0.14, rootMargin: '0px 0px -40px 0px' });
    revealEls.forEach(function (el) { io.observe(el); });
  } else {
    revealEls.forEach(function (el) { el.classList.add('in'); });
  }

  /* ---- Animated stat counters ---- */
  var counters = document.querySelectorAll('[data-count]');
  if ('IntersectionObserver' in window && counters.length) {
    var counterIO = new IntersectionObserver(function (entries) {
      entries.forEach(function (entry) {
        if (!entry.isIntersecting) return;
        var el = entry.target;
        var target = parseFloat(el.getAttribute('data-count'));
        var suffix = el.getAttribute('data-suffix') || '';
        var duration = 1400;
        var start = null;
        function step(ts) {
          if (!start) start = ts;
          var progress = Math.min((ts - start) / duration, 1);
          var eased = 1 - Math.pow(1 - progress, 3);
          var val = target * eased;
          el.textContent = (target % 1 === 0 ? Math.floor(val) : val.toFixed(1)) + suffix;
          if (progress < 1) requestAnimationFrame(step);
          else el.textContent = (target % 1 === 0 ? target : target.toFixed(1)) + suffix;
        }
        requestAnimationFrame(step);
        counterIO.unobserve(el);
      });
    }, { threshold: 0.4 });
    counters.forEach(function (el) { counterIO.observe(el); });
  }

  /* ---- Reviews slider ---- */
  var track = document.getElementById('reviewsTrack');
  var prevBtn = document.getElementById('reviewPrev');
  var nextBtn = document.getElementById('reviewNext');
  if (track && prevBtn && nextBtn) {
    function scrollAmt() {
      var card = track.querySelector('.review-card');
      return card ? card.offsetWidth + 24 : 320;
    }
    prevBtn.addEventListener('click', function () { track.scrollBy({ left: -scrollAmt(), behavior: 'smooth' }); });
    nextBtn.addEventListener('click', function () { track.scrollBy({ left: scrollAmt(), behavior: 'smooth' }); });
  }

  /* ---- Scroll-to-top ---- */
  var toTop = document.getElementById('scrollTop');
  if (toTop) {
    document.addEventListener('scroll', function () {
      if (window.scrollY > 640) toTop.classList.add('show');
      else toTop.classList.remove('show');
    }, { passive: true });
    toTop.addEventListener('click', function () {
      window.scrollTo({ top: 0, behavior: 'smooth' });
    });
  }

  /* ---- Product filter ---- */
  var filterChips = document.querySelectorAll('.filter-chip');
  var productCards = document.querySelectorAll('.product-card');
  if (filterChips.length && productCards.length) {
    filterChips.forEach(function (chip) {
      chip.addEventListener('click', function () {
        filterChips.forEach(function (c) { c.classList.remove('active'); });
        chip.classList.add('active');
        var val = chip.getAttribute('data-filter');
        productCards.forEach(function (card) {
          if (val === 'all' || card.getAttribute('data-hp') === val) {
            card.classList.remove('hidden');
          } else {
            card.classList.add('hidden');
          }
        });
      });
    });
  }

  /* ---- Simple lightbox for project images ---- */
  var lightboxTriggers = document.querySelectorAll('[data-lightbox]');
  if (lightboxTriggers.length) {
    var overlay = document.createElement('div');
    overlay.className = 'lightbox-overlay';
    overlay.innerHTML = '<button class="lightbox-close" aria-label="Close">&times;</button><div class="lightbox-inner"></div>';
    document.body.appendChild(overlay);
    var inner = overlay.querySelector('.lightbox-inner');
    var closeBtn = overlay.querySelector('.lightbox-close');

    var style = document.createElement('style');
    style.textContent = '.lightbox-overlay{position:fixed;inset:0;background:rgba(18,21,26,.94);z-index:1000;display:none;align-items:center;justify-content:center;padding:40px;}' +
      '.lightbox-overlay.open{display:flex;}' +
      '.lightbox-inner{max-width:720px;width:100%;background:#1B1F23;border:1px solid rgba(255,255,255,.12);padding:28px;}' +
      '.lightbox-close{position:absolute;top:22px;right:28px;background:none;border:none;color:#fff;font-size:32px;cursor:pointer;line-height:1;}';
    document.head.appendChild(style);

    lightboxTriggers.forEach(function (card) {
      card.addEventListener('click', function () {
        var title = card.getAttribute('data-title') || '';
        var desc = card.getAttribute('data-desc') || '';
        var svgHost = card.querySelector('.project-thumb');
        inner.innerHTML = '';
        if (svgHost) {
          var clone = svgHost.cloneNode(true);
          clone.style.aspectRatio = '16/10';
          inner.appendChild(clone);
        }
        var textWrap = document.createElement('div');
        textWrap.style.marginTop = '20px';
        textWrap.innerHTML = '<h3 style="font-family:\'Chakra Petch\',sans-serif;color:#fff;text-transform:uppercase;font-size:20px;margin:0 0 10px;">' + title + '</h3><p style="color:#AEB6BE;font-family:Inter,sans-serif;font-size:14.5px;margin:0;">' + desc + '</p>';
        inner.appendChild(textWrap);
        overlay.classList.add('open');
      });
    });
    closeBtn.addEventListener('click', function () { overlay.classList.remove('open'); });
    overlay.addEventListener('click', function (e) { if (e.target === overlay) overlay.classList.remove('open'); });
  }

});
